1 · Inventario con CIM

powershell -NoProfile -Command ^
  "Get-CimInstance Win32_ComputerSystem |
   Select-Object Manufacturer, Model, SystemType,
   @{Name='RAM_Total_GB';Expression={
     [math]::Round($_.TotalPhysicalMemory/1GB,2)
   }} | Format-List"

2 · Detección y deduplicación

function Add-Game($name, $path, $source) {
  if ([string]::IsNullOrWhiteSpace($name)) { return }
  if (!(Test-Path -LiteralPath $path)) { return }

  $cleanPath = (Resolve-Path -LiteralPath $path).Path
  $exists = $global:games | Where-Object { $_.Path -eq $cleanPath }

  if (!$exists) {
    $global:games += [PSCustomObject]@{
      Name   = $name
      Path   = $cleanPath
      Source = $source
      SizeGB = Get-FolderSizeGB $cleanPath
    }
  }
}
