El problema

Warframe mezcla Mods, Arcanos, Helminth, Fragmentos y estadísticas que cambian durante el combate. Mi objetivo fue convertir esa complejidad en una interfaz compacta, bilingüe y fácil de explicar.

Muestra de código

Aquí muestro tres fragmentos de v1.0: persistencia, sanitización/i18n y cálculo de estadísticas, sin publicar la base completa.

1 · Persistencia de builds personales

const STORAGE = "desco_nova_my_builders_v1";
let userBuilds = loadUserBuilds();

function loadUserBuilds() {
  try {
    return JSON.parse(localStorage.getItem(STORAGE) || "[]");
  } catch {
    return [];
  }
}

function saveUserBuilds() {
  localStorage.setItem(STORAGE, JSON.stringify(userBuilds));
}

2 · Nombres bilingües y salida segura

function safe(value) {
  return String(value ?? "").replace(/[&<>"']/g, ch => ({
    "&": "&amp;", "<": "&lt;", ">": "&gt;",
    '"': "&quot;", "'": "&#39;"
  }[ch]));
}

function dualName(obj) {
  const primary = lang === "es" ? obj.es : obj.en;
  const secondary = lang === "es" ? obj.en : obj.es;
  return `${safe(primary)}<span class="secondary">${safe(secondary)}</span>`;
}

3 · Estadísticas por estado

function abilityStats(build, mode) {
  let strength = 100, duration = 100,
      range = 100, efficiency = 100;

  [build.exilus, ...build.mods].forEach(name => {
    const effect = EFFECTS[name] || {};
    strength   += effect.strength   || 0;
    duration   += effect.duration   || 0;
    range      += effect.range      || 0;
    efficiency += effect.efficiency || 0;
  });

  if (mode !== "entry") {
    // Arcanos y buffs sostenibles se aplican por estado.
  }

  return { strength, duration, range,
           efficiency: Math.min(175, efficiency) };
}
